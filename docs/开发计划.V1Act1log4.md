```lean
iutNegLog_norm_embedding_sum_liesOver.{u}
    {F K : Type u} [Field F] [NumberField F]
    [Field K] [NumberField K] [Algebra F K]
    (x : F) (w : NumberField.FinitePlace F)
    (s : Finset (NumberField.FinitePlace K))
    (hs : ∀ v, v ∈ s ↔
      Ideal.LiesOver v.maximalIdeal.asIdeal w.maximalIdeal.asIdeal) :
    (∑ v ∈ s, -Real.log ‖NumberField.FinitePlace.embedding v.maximalIdeal
      (algebraMap F K x)‖) =
      (Module.finrank F K : ℝ) *
```
